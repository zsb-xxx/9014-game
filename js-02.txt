function inSafeZone(x,y){
  if(!G.shrink.active) return true;
  return x >= G.shrink.colMin && x <= G.shrink.colMax &&
         y >= G.shrink.rowMin && y <= G.shrink.rowMax;
}
function isZoneEdge(x,y){
  if(!G.shrink.active) return false;
  if(!inSafeZone(x,y)) return false;
  // 边界格
  if(x === G.shrink.colMin || x === G.shrink.colMax ||
     y === G.shrink.rowMin || y === G.shrink.rowMax) return true;
  return false;
}

/* ==================== 随机场景系统 ==================== */
let MAP = [];

function generateMap(){
  const map = Array.from({length:ROWS}, ()=>Array(COLS).fill(0));
  hubCells().forEach(([x,y])=>{map[y][x]=5;});

  // 出生区只保护近端，扩大地图后让中场更早出现复杂地形。
  const protect = [
    {x0:4,x1:10,y0:8,y1:10},   // GYC
    {x0:10,x1:14,y0:0,y1:3},   // LLY
    {x0:0,x1:4,y0:0,y1:3}      // JYM
  ];
  const protectedCell=(x,y)=>protect.some(a=>x>=a.x0&&x<=a.x1&&y>=a.y0&&y<=a.y1);
  const freeTerrain=(x,y)=>inBoard(x,y)&&map[y][x]===0&&!protectedCell(x,y);

  // 一条可穿越的曲折河带，让地图不再只是独立散点。
  let riverY=3+Math.floor(Math.random()*Math.max(1,ROWS-6));
  for(let x=0;x<COLS;x++){
    if(freeTerrain(x,riverY)&&Math.random()<0.78)map[riverY][x]=3;
    if(Math.random()<0.58){
      riverY+=Math.floor(Math.random()*3)-1;
      riverY=Math.max(3,Math.min(ROWS-4,riverY));
    }
  }

  // 中高密度基础地形：仍保留足够平原，但移动路线会明显分叉。
  for(let y=0;y<ROWS;y++){
    for(let x=0;x<COLS;x++){
      if(!freeTerrain(x,y))continue;
      const r=Math.random();
      if(r<0.10)      map[y][x]=1; // 森林
      else if(r<0.17) map[y][x]=2; // 山地
      else if(r<0.27) map[y][x]=6; // 丘陵
      else if(r<0.31) map[y][x]=4; // 城镇（仍允许整局无节点）
    }
  }

  // 追加少量森林/丘陵小片区，使地形具有“区域”而非纯噪点感。
  const patchTypes=[1,1,6,6];
  for(let p=0;p<4;p++){
    const cx=2+Math.floor(Math.random()*(COLS-4));
    const cy=2+Math.floor(Math.random()*(ROWS-4));
    const type=patchTypes[p];
    for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
      const x=cx+dx,y=cy+dy;
      if(freeTerrain(x,y)&&Math.random()<0.48)map[y][x]=type;
    }
  }
  // ZY 的独立实验室固定在边缘山地，不属于任何阵营，也不允许普通单位进入。
  const [zx,zy]=zyLabPos();map[zy][zx]=7;
  for(const [dx,dy] of [[1,0],[-1,0],[0,1],[0,-1]]){const nx=zx+dx,ny=zy+dy;if(inBoard(nx,ny)&&map[ny][nx]===0)map[ny][nx]=6;}
  if(G.zyNPC){G.zyNPC.x=zx;G.zyNPC.y=zy;}
  return map;
}

/* ==================== 临床伙伴 / 随机事件 ==================== */
function clinicalScore(fac){
  if(!G.clinical) return 0;
  const towns=Object.values(G.towns||{}).filter(x=>x===fac).length;
  const st=G.stats||{};
  const dmg=(st.damageDealt&&st.damageDealt[fac])||0;
  const kills=(st.kills&&st.kills[fac])||0;
  const boss=(st.bossDamage&&st.bossDamage[fac])||0;
  const bonus=(G.clinical.bonus&&G.clinical.bonus[fac])||0;
  return Math.max(0,Math.floor((G.round-1)*10 + dmg/18 + kills*25 + towns*20 + boss/15 + bonus));
}

function findClinicalSpawn(fac){
  const lord=G.units.find(u=>u.faction===fac&&u.hp>0&&u.type==='lord');
  const dep=DEPLOY[fac]&&DEPLOY[fac][0];
  const ax=lord?lord.x:(dep?dep[1]:Math.floor(COLS/2));
  const ay=lord?lord.y:(dep?dep[2]:Math.floor(ROWS/2));
  for(let r=1;r<=Math.max(COLS,ROWS);r++){
    for(let y=Math.max(0,ay-r);y<=Math.min(ROWS-1,ay+r);y++){
      for(let x=Math.max(0,ax-r);x<=Math.min(COLS-1,ax+r);x++){
        if(Math.max(Math.abs(x-ax),Math.abs(y-ay))!==r)continue;
        if(!isPassableTile(x,y)||unitAt(x,y))continue;
        if(G.shrink.active&&!inSafeZone(x,y))continue;
        return [x,y];
      }
    }
  }
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
    if(isPassableTile(x,y)&&!unitAt(x,y))return [x,y];
  }
  return null;
}

function checkClinicalUnlocks(fac){
  if(!G.clinical||!G.clinical.unlocked[fac])return;
  const score=clinicalScore(fac);
  CLINICAL_PARTNERS.forEach(p=>{
    if(score<p.threshold||G.clinical.unlocked[fac][p.key])return;
    if(spawnClinicalPartner(fac,p))G.clinical.unlocked[fac][p.key]=true;
  });
}
