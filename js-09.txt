  const alliesNear=G.units.filter(t=>t.hp>0&&t.faction===fac&&t.id!==u.id&&dist(t,u)<=2).length;
        if(alliesNear>=1)use=true;
      } else if(u.type==='sage'){
        const enemiesNear=targets.filter(e=>dist(e,u)<=3).length;
        if(enemiesNear>=1)use=true;
      } else use=true;
      if(use){
        const sk=SKILLS[u.type];
        u.energy-=sk.cost;u.skillCd=sk.cd+1;
        slashAt(u.x,u.y,'skill');
        if(u.type==='lord'){
          G.units.forEach(t=>{
            if(t.hp>0&&t.faction===fac&&dist(t,u)<=2){
              t.flagBuff=true;
              const heal=30;
              t.hp=Math.min(t.maxHp,t.hp+heal);
              float(t,heal,'heal');
            }
          });
          log(`<b>${factionName(fac)}</b> 统帅发动战旗号令`,'skill');
        } else if(u.type==='brave'){
          u.rageBuff=true;
          log(`<b>${factionName(fac)}</b> 猛将发动狂暴突袭`,'skill');
        } else if(u.type==='sage'){
          let cnt=0;
          G.units.forEach(t=>{
            if(t.hp<=0)return;
            if(t.faction===fac)return;
            if(dist(t,u)>3)return;
            if(G.bossActive && !G.bossDefeated && !(t.isBoss||t.isMinion))return;
            if(!G.bossActive && (t.isBoss||t.isMinion))return;
            const skillDmg=mitigateBossDamage(t,30);t.hp-=skillDmg;float(t,skillDmg,'crit');shakeUnit(t);cnt++;
            if(G.stats.damageDealt[fac]!==undefined)G.stats.damageDealt[fac]+=skillDmg;
            if(t.isBoss&&G.stats.bossDamage[fac]!==undefined)G.stats.bossDamage[fac]+=skillDmg;
            recordMXYBalanceDamage(t,fac,skillDmg);
            if(t.hp<=0){log(`<b>${factionName(t.faction)} · ${t.typeName}</b> 阵亡`,'kill');
              if(G.stats.unitsLost[t.faction]!==undefined)G.stats.unitsLost[t.faction]++;
              if(G.stats.kills[fac]!==undefined)G.stats.kills[fac]++;
              recordRebelHead(t,fac);}
          });
          log(`<b>${factionName(fac)}</b> 谋士发动火计焚营，命中 ${cnt} 目标`,'skill');
        } else if(u.type==='guard'){
          u.shieldWall=true;
          u.hp=Math.min(u.maxHp,u.hp+20);
          float(u,20,'heal');
          log(`<b>${factionName(fac)}</b> 精兵发动铁壁结阵`,'skill');
        }
        u.acted=true;u.moved=true;
        renderUnits();renderHUD();
        setTimeout(step,700);return;
      }
    }

    targets.sort((a,b)=>{
      const bal=G.mxy&&G.mxy.balance&&G.mxy.balance.active?G.mxy.balance.target:null;
      const kA=(a.hp<=calcAtk(u)*1.5?-100:0)+dist(u,a)*(counterState(u,a)==='adv'?0.6:1)+(a.isMXY&&a.envoy?35:0)+(bal&&a.faction===bal&&fac!==bal?-24:0);
      const kB=(b.hp<=calcAtk(u)*1.5?-100:0)+dist(u,b)*(counterState(u,b)==='adv'?0.6:1)+(b.isMXY&&b.envoy?35:0)+(bal&&b.faction===bal&&fac!==bal?-24:0);
      return kA-kB;
    });
    const target=targets[0];
    if(!target){u.acted=true;renderUnits();setTimeout(step,180);return;}

    if(inPattern(u,target.x,target.y)){
      const flank=flankBonus(u,target);
      doAttack(u,target,flank?1.5:1);
      setTimeout(step,640);return;
    }
    const moves=computeMoves(u);
    if(!moves.length){u.acted=true;renderUnits();setTimeout(step,180);return;}
    let best=null,bestScore=Infinity;
    for(const [mx,my] of moves){
      let score=Math.abs(mx-target.x)+Math.abs(my-target.y);
      const testU={...u,x:mx,y:my};
      if(inPattern(testU,target.x,target.y))score-=60;
      if(MAP[my][mx]===4&&G.towns[mx+','+my]!==fac)score-=20;
      if(MAP[my][mx]===5)score-=5;
      if(G.disaster){
        const k=G.disaster.key;
        if(k==='fire'&&MAP[my][mx]===1)score+=25;
        if(k==='blizzard'&&MAP[my][mx]!==5)score+=8;
        if(k==='flood'&&MAP[my][mx]===3)score+=15;
      }
      // 安全区外重罚
      if(G.shrink.active && !inSafeZone(mx,my)) score += 60;
      // 收缩期倾向中心
      if(G.shrink.active && G.round >= G.shrink.nextRound - 1){
        const cx=(G.shrink.colMin+G.shrink.colMax)/2;
        const cy=(G.shrink.rowMin+G.shrink.rowMax)/2;
        score += Math.abs(mx-cx)+Math.abs(my-cy);
      }
      targets.forEach(e=>{if(e!==target&&Math.abs(mx-e.x)+Math.abs(my-e.y)<=1)score+=6;});
      if(score<bestScore){bestScore=score;best=[mx,my];}
    }
    if(best&&(best[0]!==u.x||best[1]!==u.y)){
      u.x=best[0];u.y=best[1];u.moved=true;
      if(MAP[u.y][u.x]===4){
        const k=u.x+','+u.y;
        if(G.towns[k]!==u.faction){G.towns[k]=u.faction;log(`<b>${factionName(u.faction)}</b> 接管算力节点 (${u.x},${u.y})`,'evt');}
      }
      renderUnits();refreshTiles();
    }
    setTimeout(()=>{
      if(u.hp<=0){step();return;}
      let t2pool=alive().filter(e=>e.faction!==fac);
      if(G.bossActive && !G.bossDefeated){
        const bossUnits=t2pool.filter(e=>e.isBoss||e.isMinion);
        if(bossUnits.length) t2pool=bossUnits;
      }
      const bal=G.mxy&&G.mxy.balance&&G.mxy.balance.active?G.mxy.balance.target:null;
      const t2=t2pool.sort((a,b)=>(dist(u,a)+(a.isMXY&&a.envoy?6:0)+(bal&&a.faction===bal&&fac!==bal?-4:0))-(dist(u,b)+(b.isMXY&&b.envoy?6:0)+(bal&&b.faction===bal&&fac!==bal?-4:0)))[0];
      if(t2&&inPattern(u,t2.x,t2.y)){
        const flank=flankBonus(u,t2);
        doAttack(u,t2,flank?1.5:1);
      } else {u.acted=true;renderUnits();}
      setTimeout(step,580);
    },420);
  }
  step();
}

/* ==================== BOSS 讨伐结算 ==================== */
function resolveBossRewards(){
  const scores=MAJOR_FACTIONS.map(f=>({f,dmg:(G.stats.bossDamage&&G.stats.bossDamage[f])||0})).sort((a,b)=>b.dmg-a.dmg);
  const top=scores[0]&&scores[0].dmg>0?scores[0]:null;
  scores.forEach(({f,dmg})=>{
    if(dmg<=0)return;
    G.gold[f]=(G.gold[f]||0)+20;
    G.units.filter(u=>u.hp>0&&u.faction===f&&PRINCE_CORE_TYPES.has(u.type)).forEach(u=>gainExp(u,15));
  });
  if(top){
    G.gold[top.f]=(G.gold[top.f]||0)+50;
    G.units.filter(u=>u.hp>0&&u.faction===top.f&&PRINCE_CORE_TYPES.has(u.type)).forEach(u=>{
      const heal=Math.max(1,Math.round(u.maxHp*0.25));u.hp=Math.min(u.maxHp,u.hp+heal);float(u,heal,'heal');gainExp(u,25);
    });
    log(`✦ <b>讨伐魁首：${factionName(top.f)}</b> · 对 BOSS 造成 ${top.dmg} 伤害，额外获得 50 金、全军整备与经验。`,'boss');
  } else log(`✦ BOSS 已陨落，但三方未形成可统计的讨伐贡献。`,'boss');
  scores.filter(x=>x.dmg>0&&(!top||x.f!==top.f)).forEach(x=>log(`✦ ${factionName(x.f)} 参与讨伐：${x.dmg} 伤害，获得 20 金与经验。`,'boss'));
  G.bossRewards={top:top?top.f:null,scores};
  G.units.filter(u=>u.isMinion&&u.hp>0).forEach(u=>u.hp=0);
  boardEl.classList.remove('boss-zone');
  renderBossBar();renderUnits();renderHUD();
}

/* ==================== 胜负 ==================== */
function checkWin(){
  if(G.over)return;
  checkMXYDefeat();
  if(G.player==='MXY'&&G.mxy&&G.mxy.defeated){G.over=true;G.busy=false;renderHUD();endGame('mxy-defeat','MXY');return;}
  if(G.mxy&&G.mxy.won){G.over=true;G.busy=false;renderHUD();endGame('mxy-victory','MXY');return;}
  const majorAlive=MAJOR_FACTIONS.filter(f=>G.units.some(u=>u.hp>0&&u.faction===f));
  if(G.player==='MXY'&&G.mxy&&G.mxy.spawned&&!G.mxy.won&&majorAlive.length<=1){
    G.over=true;G.busy=false;renderHUD();endGame('mxy-collapse','MXY');return;
  }
  if(!G.bossActive){
    if(G.roleVictory&&G.roleVictory.faction&&G.units.some(u=>u.hp>0&&u.faction===G.roleVictory.faction)){
      G.over=true;G.busy=false;renderHUD();const w=G.roleVictory.faction;endGame(w===G.player?'win':'lose',w);return;
    }
  }
  if(G.bossActive&&!G.bossDefeated){
    const boss=G.units.find(u=>u.isBoss&&u.hp>0);
    if(!boss){
      G.bossDefeated=true;G.bossActive=false;G.busy=false;
      resolveBossRewards();
      log(`✦ 联合讨伐结束。BOSS 已陨落，但天下之争继续。`,'boss');
      renderHUD();
      // BOSS 不再是直接胜利条件；恢复各身份的终局进度判断。
      updateRebelEndgame();
      return;
    }
    const playerAlive=G.units.some(u=>u.faction===G.player&&u.hp>0);
    if(!playerAlive){G.over=true;G.busy=false;renderHUD();endGame('boss-defeat');return;}
    return;
  }
  const facs=G.order.filter(k=>G.units.some(u=>u.faction===k&&u.hp>0));
  if(facs.length<=1&&!G.over){
    G.over=true;G.busy=false;renderHUD();
    const winner=facs[0];
    if(winner){const isPlayer=winner===G.player;endGame(isPlayer?'win':'lose',winner);}else endGame('draw');
  }
}
