(function enhanceUX(){
  const glow=document.createElement('div');
  glow.className='cursor-glow';
  document.body.appendChild(glow);
  let mx=0,my=0,cx=0,cy=0;
  document.addEventListener('mousemove',e=>{
    mx=e.clientX;my=e.clientY;
    glow.classList.add('on');
    const t=e.target.closest('.fcard,.card,.btn,.abtn,.unit');
    if(t){
      const c=getComputedStyle(t).getPropertyValue('--c')||'#d4af37';
      glow.style.background=`radial-gradient(circle,${c}22,${c}08 40%,transparent 70%)`;
    } else {
      glow.style.background='radial-gradient(circle,rgba(212,175,55,.14),rgba(212,175,55,.04) 40%,transparent 70%)';
    }
  });
  document.addEventListener('mouseleave',()=>glow.classList.remove('on'));
  (function tick(){
    cx+=(mx-cx)*.16;cy+=(my-cy)*.16;
    glow.style.left=cx+'px';glow.style.top=cy+'px';
    requestAnimationFrame(tick);
  })();

  document.addEventListener('pointerdown',e=>{
    const btn=e.target.closest('.btn,.abtn,.help-btn,.fcard,.card');
    if(!btn) return;
    const r=btn.getBoundingClientRect();
    const size=Math.max(r.width,r.height);
    const ripple=document.createElement('span');
    ripple.className='ripple';
    ripple.style.width=ripple.style.height=size+'px';
    ripple.style.left=(e.clientX-r.left-size/2)+'px';
    ripple.style.top =(e.clientY-r.top -size/2)+'px';
    const old=btn.querySelector('.ripple');
    if(old) old.remove();
    btn.appendChild(ripple);
    setTimeout(()=>ripple.remove(),700);
  });

  document.addEventListener('mousemove',e=>{
    const card=e.target.closest('.fcard');
    if(!card) return;
    const r=card.getBoundingClientRect();
    const px=(e.clientX-r.left)/r.width-.5;
    const py=(e.clientY-r.top )/r.height-.5;
    card.style.transform=`translateY(-10px) rotateX(${-py*10}deg) rotateY(${px*10}deg)`;
  });
  document.addEventListener('mouseout',e=>{
    const card=e.target.closest('.fcard');
    if(!card) return;
    if(card.contains(e.relatedTarget)) return;
    card.style.transform='';
  });

  let audioCtx;
  const beep=(freq=720,dur=0.06,vol=.04)=>{
    try{
      if(!audioCtx) audioCtx=new (window.AudioContext||window.webkitAudioContext)();
      const o=audioCtx.createOscillator(),g=audioCtx.createGain();
      o.type='sine';o.frequency.value=freq;
      g.gain.setValueAtTime(vol,audioCtx.currentTime);
      g.gain.exponentialRampToValueAtTime(.0001,audioCtx.currentTime+dur);
      o.connect(g).connect(audioCtx.destination);
      o.start();o.stop(audioCtx.currentTime+dur);
    }catch(e){}
  };
  document.addEventListener('pointerdown',e=>{
    if(e.target.closest('.btn,.abtn'))  beep(680,.05,.035);
    if(e.target.closest('.card'))       beep(880,.06,.04);
    if(e.target.closest('.unit'))       beep(520,.04,.03);
  },{passive:true});
})();

(function initHudTabsAndCollapse(){
  function activateTab(shell,target,remember=true){
    if(!shell||!target)return;
    shell.querySelectorAll(':scope > .hud-tabs .hud-tab').forEach(btn=>{
      const on=btn.dataset.tabTarget===target;
      btn.classList.toggle('active',on);
      btn.setAttribute('aria-selected',on?'true':'false');
      if(on)btn.classList.remove('has-alert');
    });
    shell.querySelectorAll(':scope > .hud-tab-pane').forEach(pane=>pane.classList.toggle('active',pane.dataset.tabPane===target));
    if(remember){
      try{localStorage.setItem('9014-hud-'+shell.dataset.tabGroup,target);}catch(e){}
    }
  }

  document.querySelectorAll('.hud-tab-shell').forEach(shell=>{
    shell.querySelectorAll(':scope > .hud-tabs .hud-tab').forEach(btn=>{
      btn.addEventListener('click',()=>activateTab(shell,btn.dataset.tabTarget));
    });
    let saved='';
    try{saved=localStorage.getItem('9014-hud-'+shell.dataset.tabGroup)||'';}catch(e){}
    const valid=saved&&shell.querySelector(`:scope > .hud-tabs .hud-tab[data-tab-target="${saved}"]`);
    if(valid)activateTab(shell,saved,false);
  });

  document.querySelectorAll('.collapsible-panel>.panel-h').forEach(head=>{
    head.setAttribute('title','点击折叠 / 展开');
    head.setAttribute('role','button');
    head.setAttribute('tabindex','0');
    head.setAttribute('aria-expanded','true');
    const toggle=()=>{
      const panel=head.parentElement;
      const collapsed=panel.classList.toggle('is-collapsed');
      head.setAttribute('aria-expanded',collapsed?'false':'true');
    };
    head.addEventListener('click',toggle);
    head.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' '){e.preventDefault();toggle();}});
  });

  // 战报在后台增长时，用红点提示，但不强制抢走玩家当前标签页。
  const logNode=document.getElementById('log');
  const rightShell=document.querySelector('.hud-tab-shell[data-tab-group="right"]');
  const reportBtn=rightShell?.querySelector('.hud-tab[data-tab-target="report"]');
  if(logNode&&reportBtn){
    new MutationObserver(()=>{
      if(!reportBtn.classList.contains('active'))reportBtn.classList.add('has-alert');
    }).observe(logNode,{childList:true});
  }

  // 天灾/收圈真正激活后，在“战局”标签上给提示点；MXY现身后提示“势力”标签。
  const leftShell=document.querySelector('.hud-tab-shell[data-tab-group="left"]');

    const battleBtn=leftShell.querySelector('.hud-tab[data-tab-target="battle"]');
    const factionBtn=leftShell.querySelector('.hud-tab[data-tab-target="faction"]');
    const syncAlerts=()=>{
      const hazard=document.getElementById('disasterPanel')?.classList.contains('active')||document.getElementById('zonePanel')?.classList.contains('active');
      if(hazard&&!battleBtn.classList.contains('active'))battleBtn.classList.add('has-alert');
      const mxyPanel=document.querySelector('.mxy-panel');
      const mxyVisible=mxyPanel&&getComputedStyle(mxyPanel).display!=='none';
      if(mxyVisible&&!factionBtn.classList.contains('active'))factionBtn.classList.add('has-alert');
    };
    const watch=new MutationObserver(syncAlerts);
    const dis=document.getElementById('disasterPanel'),zone=document.getElementById('zonePanel'),mxy=document.querySelector('.mxy-panel');
    if(dis)watch.observe(dis,{attributes:true,attributeFilter:['class']});
    if(zone)watch.observe(zone,{attributes:true,attributeFilter:['class']});
