'di-fire');
    if(key==='flood'){
      let nearRiver=MAP[y][x]===3;
      if(!nearRiver){for(const [dx,dy] of DIRS){const nx=x+dx,ny=y+dy;if(inBoard(nx,ny)&&MAP[ny][nx]===3){nearRiver=true;break;}}}
      if(nearRiver)el.classList.add('di-flood');
    }
    if(key==='blizzard'&&!onZY(x,y))el.classList.add('di-blizzard');
    if(key==='plague')el.classList.add('di-plague');
    if(key==='zy-surge'&&onZY(x,y))el.classList.add('di-zy-surge');
    if(key==='storm')el.classList.add('di-storm');
  });
  boardEl.classList.toggle('di-dark',key==='storm'||key==='bloodmoon'||key==='plague');
}

function buildUnits(){
  G.units=[];G.unitSeq=0;G.spies={};
  Object.keys(DEPLOY).forEach(fac=>{
    DEPLOY[fac].forEach(([type,x,y])=>{
      const t=TYPES[type];
      const u={
        id:'u'+(G.unitSeq++),faction:fac,type,
        short:t.short,typeName:t.name,x,y,
        hp:t.hp,maxHp:t.hp,atk:t.atk,mov:t.mov,
        moved:false,acted:false,buff:1,movBonus:0,shield:0,
        energy:50,level:1,exp:0,skillCd:0,
        rageBuff:false,flagBuff:false,shieldWall:false,zyBuff:false,slow:false
      };
      G.units.push(u);
    });
  });
  const facs=['GYC','LLY','JYM'];
  facs.forEach(fac=>{
    const ownUnits=G.units.filter(u=>u.faction===fac&&u.type!=='lord');
    if(!ownUnits.length)return;
    const spy=ownUnits[Math.floor(Math.random()*ownUnits.length)];
    const others=facs.filter(f=>f!==fac);
    const realFac=others[Math.floor(Math.random()*others.length)];
    G.spies[spy.id]={realFaction:realFac};
  });
}

function victoryStatusText(fac){
  if(fac==='GYC'){
    const heads=(G.rebelHeads?.LLY?1:0)+(G.rebelHeads?.JYM?1:0);

    return `特殊胜利：亲自击破核心 ${heads}/2`;
  }
  if(fac==='LLY'){
    const townExists=mapHasTown(),hasTown=Object.values(G.towns||{}).some(v=>v==='LLY');
    const zy=G.units.filter(u=>u.hp>0&&u.faction==='LLY'&&PRINCE_CORE_TYPES.has(u.type)&&onZY(u.x,u.y)).length;

  }
  if(fac==='JYM'){
    const regs=jymControlledRegions();
    const zy=G.units.filter(u=>u.hp>0&&u.faction==='JYM'&&PRINCE_CORE_TYPES.has(u.type)&&onZY(u.x,u.y)).length;

  return '';
}
