function mxyVerdictStatus(m){

  if(currentMXYEvent()?.key!=='verdict')return {ok:false,reason:'尚未进入裁局阶段',fac:null,rep:null};
  if(aliveMaj.length<2)return {ok:false,reason:'天下已只剩一方，领主失去制衡空间',fac:null,rep:null};
  const preferred=aliveMaj.filter(f=>f!==G.mxy.lastBalanceTarget);
  const pool=(preferred.length?preferred:aliveMaj).map(f=>({fac:f,rep:mxyRepresentative(f)})).filter(x=>x.rep);
  if(!pool.length)return {ok:false,reason:'当前无可裁局代表',fac:null,rep:null};
  const adjacent=pool.filter(x=>cheb(x.rep,m)<=1).sort((a,b)=>dist(m,a.rep)-dist(m,b.rep));
  if(adjacent.length)return {ok:true,fac:adjacent[0].fac,rep:adjacent[0].rep,reason:'已抵达裁局代表身边'};
  const nearest=pool.sort((a,b)=>dist(m,a.rep)-dist(m,b.rep))[0];
  return {ok:false,fac:nearest.fac,rep:nearest.rep,reason:`需接近 ${factionName(nearest.fac)} 的代表完成最终裁局`};

      if(fs.length>=2)cells.push([x,y]);
    }
    if(cells.length)return cells.sort((a,b)=>Math.abs(a[0]-m.x)+Math.abs(a[1]-m.y)-Math.abs(b[0]-m.x)-Math.abs(b[1]-m.y))[0];
  }
  if(ev&&ev.key==='balance'){const t=mxyStrongestFaction();const rep=t?mxyRepresentative(t):null;if(rep)return [rep.x,rep.y];}
  if(ev&&ev.key==='verdict'){const st=mxyVerdictStatus(m);if(st.rep)return [st.rep.x,st.rep.y];}
  return [Math.floor(COLS/2),Math.floor(ROWS/2)];
}
