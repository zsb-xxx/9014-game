function computeMoves(u){
  const res=[];
  const seen=new Map();
  seen.set(u.x+','+u.y,0);
  const q=[[u.x,u.y,0]];
  const maxStep=calcMov(u);
  while(q.length){
    const [x,y,c]=q.shift();
    if(c>=maxStep)continue;
    for(const [dx,dy] of DIRS){
      const nx=x+dx,ny=y+dy;
      if(!inBoard(nx,ny))continue;
      if(!isPassableTile(nx,ny))continue;
      const nc=c+terrainAt(nx,ny).movCost;
      if(nc>maxStep)continue;
      const k=nx+','+ny;
      if(seen.has(k)&&seen.get(k)<=nc)continue;
      seen.set(k,nc);
      const occ=unitAt(nx,ny);
      if(occ&&occ.faction!==u.faction)continue;
      q.push([nx,ny,nc]);
      if(!occ)res.push([nx,ny]);
    }
  }
  return res;
}

function gainExp(u,amount){
  if(u.isBoss||u.isMinion||u.isMXY)return;
  u.exp+=amount;
  float(u,amount,'exp');
  const idx=Math.min(u.level,LEVEL_EXP.length-1);
  if(u.exp>=LEVEL_EXP[idx]&&u.level<6){
    u.level++;
    u.maxHp+=12;u.hp=Math.min(u.maxHp,u.hp+12);
    u.atk+=3;
    if(u.level===4)u.mov+=1;
    lvFX(u.x,u.y);
    log(`<b>${factionName(u.faction)} · ${u.typeName}</b> 升至 Lv${u.level}！`,'lv');
  }
}


const DUEL_CORE_TYPES=new Set(['lord','brave','sage','guard']);

function duelPower(u,opponent){
  // 单挑强调个体战力：攻击约占一半，其余来自兵力底盘、等级和当前状态。
  const atkPart=calcAtk(u)*2.15;
  const bodyPart=u.maxHp*0.24;
  const levelPart=(u.level||1)*11;
  const hpRatio=Math.max(0,Math.min(1,u.hp/u.maxHp));
  const conditionPart=18+hpRatio*24;
  let p=atkPart+bodyPart+levelPart+conditionPart;
  const ct=counterState(u,opponent);
  if(ct==='adv')p*=1.08;
  else if(ct==='dis')p*=0.94;
  return Math.max(1,Math.round(p));
}

function canTriggerDuel(att,def){
  if(!G.duel||G.round<2||G.duel.lastRound===G.round)return false;
  if(!att||!def||att.hp<=0||def.hp<=0||att.faction===def.faction)return false;
  if(dist(att,def)!==1)return false;
  if(att.isBoss||def.isBoss||att.isMinion||def.isMinion)return false;
  if(!DUEL_CORE_TYPES.has(att.type)||!DUEL_CORE_TYPES.has(def.type))return false;
  return Math.random()<(G.duel.chance||0.07);
}

function resolveDuel(att,def){
  G.duel.lastRound=G.round;G.duel.count++;
  const pA=duelPower(att,def),pB=duelPower(def,att);
  // 只保留 ±10% 的临阵随机，保证绝大多数结果仍由基础战力决定。
  const rA=Math.round(pA*(0.90+Math.random()*0.20));
  const rB=Math.round(pB*(0.90+Math.random()*0.20));
  let winner,loser,winRoll,loseRoll;
  if(rA===rB){
    if(pA>=pB){winner=att;loser=def;winRoll=rA;loseRoll=rB;}
    else{winner=def;loser=att;winRoll=rB;loseRoll=rA;}
  }else if(rA>rB){winner=att;loser=def;winRoll=rA;loseRoll=rB;}
  else{winner=def;loser=att;winRoll=rB;loseRoll=rA;}

  const gap=Math.max(0,Math.min(1,(winRoll-loseRoll)/Math.max(1,winRoll)));
  const loserPct=0.42+Math.min(0.28,gap*0.85);
  const winnerPct=Math.max(0.10,0.20-gap*0.18);
  const loserDmg=Math.max(1,Math.round(loser.maxHp*loserPct));
  let winnerDmg=Math.max(1,Math.round(winner.maxHp*winnerPct));
  // 单挑胜者至少保留 1 点兵力，避免出现“胜者同时阵亡”的含糊结算。
  winnerDmg=Math.min(winnerDmg,Math.max(0,winner.hp-1));

  loser.hp-=loserDmg;winner.hp-=winnerDmg;
  float(loser,loserDmg,'crit');float(winner,winnerDmg,'');
  shakeUnit(loser);shakeUnit(winner);slashAt(att.x,att.y);slashAt(def.x,def.y);

  if(G.stats.damageDealt[winner.faction]!==undefined)G.stats.damageDealt[winner.faction]+=loserDmg;
  if(G.stats.damageDealt[loser.faction]!==undefined)G.stats.damageDealt[loser.faction]+=winnerDmg;
  recordMXYBalanceDamage(loser,winner.faction,loserDmg);
  recordMXYBalanceDamage(winner,loser.faction,winnerDmg);
  if(G.duel.wins[winner.faction]!==undefined)G.duel.wins[winner.faction]++;
  if(G.duel.fights){
    if(G.duel.fights[att.faction]!==undefined)G.duel.fights[att.faction]++;
    if(G.duel.fights[def.faction]!==undefined)G.duel.fights[def.faction]++;
  }

  log(`⚔ <b>阵前单挑！</b> ${factionName(att.faction)} · ${att.typeName}（战力 ${pA}） VS ${factionName(def.faction)} · ${def.typeName}（战力 ${pB}）`,'duel');
  log(`单挑临阵值 <b>${rA} : ${rB}</b>，<b>${factionName(winner.faction)} · ${winner.typeName}</b> 胜出`,'duel');

  gainExp(winner,loser.hp<=0?70:35);
  if(loser.hp>0)gainExp(loser,10);
  if(loser.hp<=0){
    log(`<b>${factionName(loser.faction)} · ${loser.typeName}</b> 单挑败北，阵亡！`,'kill');
    if(G.stats.unitsLost[loser.faction]!==undefined)G.stats.unitsLost[loser.faction]++;
    if(G.stats.kills[winner.faction]!==undefined)G.stats.kills[winner.faction]++;
    recordRebelHead(loser,winner.faction);
  }

  showDuelResult(att,def,pA,pB,rA,rB,winner,loser,loserDmg,winnerDmg);
  att.acted=true;att.moved=true;
  renderUnits();renderHUD();
  setTimeout(checkWin,620);
}

function doAttack(att,def,mulOverride){
  if(canTriggerDuel(att,def)){resolveDuel(att,def);return;}
  slashAt(def.x,def.y);
  const raging=isRaging(att)||att.rageBuff;
  let baseMul=mulOverride!==undefined?mulOverride:1;
  const ctState=counterState(att,def);
  let ctMul=1;
  if(ctState==='adv')ctMul=1.5;else if(ctState==='dis')ctMul=0.75;
  let dmg=Math.round(calcAtk(att)*baseMul*ctMul);
  const crit=Math.random()<0.15;
  if(crit)dmg=Math.round(dmg*1.6);
  if(def.type==='guard')dmg=Math.round(dmg*0.85);
  if(def.shieldWall)dmg=Math.round(dmg*0.5);
  dmg=Math.round(dmg/terrainAt(def.x,def.y).def);
  if(def.rageBuff)dmg=Math.round(dmg*1.2);
  dmg=mitigateBossDamage(def,dmg);
  if(def.isMXY&&def.envoy)dmg=Math.max(1,Math.round(dmg*0.55));

  // LBL 隐身采用单独闪避池：隐身期间不再叠加地形闪避。
  // 若本次没有闪掉，则隐身立即被识破并由这一击正常造成伤害。
  let brokeInvisibility=false;
  if(def.invisible){
    if(Math.random()<(def.invisibleChance||0.35)){
      log(`${def.typeName} 隐身闪避了攻击`,'sys');
      att.acted=true;att.moved=true;
      renderUnits();renderHUD();
      setTimeout(checkWin,400);return;
    }
    def.invisible=false;
    def.invisibleChance=0;
    brokeInvisibility=true;
    if(def.isBoss&&def.faction==='LBL')log(`☯ LBL 的隐身被识破！`,'boss');
  }
  if(!brokeInvisibility && Math.random()<terrainAt(def.x,def.y).eva){
    log(`${def.typeName} 闪避了攻击`,'sys');
    att.acted=true;att.moved=true;
    renderUnits();renderHUD();
    setTimeout(checkWin,400);return;
  }
  let absorbed=0;
  if(def.shield>0){
    absorbed=Math.min(def.shield,dmg);
    def.shield-=absorbed;dmg-=absorbed;
    if(absorbed>0)float(def,absorbed,'shield');
  }
  const dealt=Math.max(0,dmg);
  if(dealt>0){
    def.hp-=dealt;
    const kind=crit?'crit':(ctState==='adv'?'adv':(raging?'rage':''));
    float(def,dealt,kind);
  }
  shakeUnit(def);
  if(G.stats.damageDealt[att.faction]!==undefined)G.stats.damageDealt[att.faction]+=dealt;
  if(def.isBoss&&G.stats.bossDamage[att.faction]!==undefined)G.stats.bossDamage[att.faction]+=dealt;
  recordMXYBalanceDamage(def,att.faction,dealt);

  const ctLabel=ctState==='adv'?' <span style="color:#ffd166">[克制]</span>':ctState==='dis'?' <span style="color:#8d8578">[被克]</span>':'';
  const rageTag=raging?' <span style="color:#ff8574">[狂暴]</span>':'';
  log(`<span style="color:${factionColor(att.faction)}">${factionName(att.faction)}</span> ${att.typeName}${rageTag}${ctLabel} 攻击 <span style="color:${factionColor(def.faction)}">${factionName(def.faction)}</span> ${def.typeName}，造成 <b>${dealt}</b> 伤害${crit?'（暴击）':''}`,'hit');

  if(def.hp>0&&dist(att,def)===1&&def.type!=='sage'&&!att.isBoss&&!att.isMinion&&!(def.isMXY&&def.envoy)){
    const back=Math.round(calcAtk(def)*0.5);
    att.hp-=back;shakeUnit(att);float(att,back,'');
    log(`${def.typeName} 反击，造成 ${back} 伤害`,'hit');
    if(att.hp<=0){
      log(`<b>${factionName(att.faction)} · ${att.typeName}</b> 阵亡`,'kill');
      if(G.stats.unitsLost[att.faction]!==undefined)G.stats.unitsLost[att.faction]++;
      recordRebelHead(att,def.faction);
    }
  }
  gainExp(att,15);
  if(def.hp<=0){
    gainExp(att,60);
    log(`<b>${factionName(def.faction)} · ${def.typeName}</b> 阵亡`,'kill');
    if(G.stats.unitsLost[def.faction]!==undefined)G.stats.unitsLost[def.faction]++;
    if(G.stats.kills[att.faction]!==undefined)G.stats.kills[att.faction]++;
    recordRebelHead(def,att.faction);
  }
  att.acted=true;att.moved=true;
  renderUnits();renderHUD();
  setTimeout(checkWin,520);
}

function flankBonus(att,def){
  const dx=att.x-def.x,dy=att.y-def.y;
  const dirs=[[1,0],[-1,0],[0,1],[0,-1]];
  for(const [ddx,ddy] of dirs){
    if(dx===ddx&&dy===ddy){
      const fx=def.x-ddx,fy=def.y-ddy;
      if(!inBoard(fx,fy))return false;
      const ally=unitAt(fx,fy);
      return ally&&ally.faction===att.faction&&ally.id!==att.id;
    }
  }
  return false;
}

function enterSkillMode(){
  const u=G.selected;
  if(!u||!canUseSkill(u))return;
  if(u.type==='brave'||u.type==='guard'||u.type==='mxy_lord'){castSkill();return;}
  G.skillMode=true;
  G.moveCells=[];G.atkCells=[];
  const cells=[];
  if(u.type==='lord'){
    for(let dy=-2;dy<=2;dy++)for(let dx=-2;dx<=2;dx++){
      if(dx===0&&dy===0)continue;
      if(Math.abs(dx)+Math.abs(dy)>2)continue;
      const nx=u.x+dx,ny=u.y+dy;
      if(!inBoard(nx,ny))continue;
      cells.push([nx,ny]);
    }
  } else if(u.type==='sage'){
    for(let dy=-3;dy<=3;dy++)for(let dx=-3;dx<=3;dx++){
      if(dx===0&&dy===0)continue;
      if(Math.abs(dx)+Math.abs(dy)>3)continue;
      const nx=u.x+dx,ny=u.y+dy;
      if(!inBoard(nx,ny))continue;
      cells.push([nx,ny]);
    }
  }
  G.skillCells=cells;
  renderHighlights();
