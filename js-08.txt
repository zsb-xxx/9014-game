function bossTurn(onDone){
  const boss=G.units.find(u=>u.isBoss&&u.hp>0);
  if(!boss){onDone&&onDone();return;}
  if(!G.bossKey){onDone&&onDone();return;}
  const b=BOSSES[G.bossKey];
  // LBL 的隐身最多持续一个完整 BOSS 周期；若期间始终未被命中，下一轮行动前自动显形。
  if(G.bossKey==='LBL'&&boss.invisible){
    boss.invisible=false;
    boss.invisibleChance=0;
    log(`☯ LBL 的混沌隐匿消散，重新显形`,'boss');
  }
  const phase=updateBossPhase(boss,true);
  const regenRate=phase>=3?BOSS_RULES.regenFinal:BOSS_RULES.regenNormal;
  const regen=Math.round(boss.maxHp*regenRate);
  boss.hp=Math.min(boss.maxHp,boss.hp+regen);
  float(boss,regen,'heal');
  log(`${b.icon} ${b.name} 汲取本源之力，回复 ${regen} 兵力 · ${bossPhaseLabel(phase)}`,'boss');
  renderUnits();renderBossBar();

  const actions=phase>=3?2:1;
  let n=0;
  const run=()=>{
    if(G.over||boss.hp<=0){onDone&&onDone();return;}
    if(n>=actions){onDone&&onDone();return;}
    n++;
    if(actions>1)log(`☠ ${b.name} 最终阶段行动 ${n}/${actions}`,'boss');
    setTimeout(()=>bossActOnce(boss,()=>{
      updateBossPhase(boss,true);
      renderBossBar();
      run();
    }),n===1?500:260);
  };
  run();
}
function bossTurnXZR(boss,onDone){

  if(!enemies.length){onDone&&onDone();return;}
  const phase=updateBossPhase(boss,false);
  const suppression=phase>=2?enemies:enemies.filter(e=>cheb(e,boss)<=3);
  suppression.forEach(t=>{
    if(t.bossSuppressedRound!==G.round){
      t.buff=(t.buff||1)*0.8;
      t.slow=true;
      t.bossSuppressedRound=G.round;
    }
  });

  const hitCount=phase>=2?2:1;
  const pool=(phase>=2?enemies:suppression).filter(t=>t.hp>0);
  const picks=pool.slice().sort(()=>Math.random()-0.5).slice(0,hitCount);
  const atk=bossEffectiveAtk(boss);

    }
    renderUnits();renderHUD();
  },i*220+250));
  setTimeout(()=>{
    renderUnits();renderHUD();renderBossBar();checkWin();
    onDone&&onDone();
  },900+picks.length*220);
}

function checkSpyReveal(){
  if(G.over)return;
  if(G.round<4)return;
  const base = G.scenario ? G.scenario.spyBaseRate : 0.15;
  let rate = base;
  if(G.round>=8) rate = base * 2;
  Object.keys(G.spies).forEach(uid=>{
    if(G.spies[uid].revealed)return;
    if(Math.random()<rate) revealSpy(uid);
  });
}
function revealSpy(uid){
  const u=G.units.find(x=>x.id===uid&&x.hp>0);
  if(!u)return;
  const info=G.spies[uid];
  if(!info||info.revealed)return;
  info.revealed=true;
  const oldFac=u.faction;
  const newFac=info.realFaction;
  u.faction=newFac;
  u.spyRevealed=true;
  G.stats.spyRevealed=G.stats.spyRevealed||0;
  G.stats.spyRevealed++;

  let hit=0;
  for(const [dx,dy] of DIRS){
    const nx=u.x+dx,ny=u.y+dy;
    if(!inBoard(nx,ny))continue;
    const t=unitAt(nx,ny);
    if(t&&t.faction===oldFac){

      hit++;
      if(t.hp<=0){
        log(`<b>${factionName(t.faction)} · ${t.typeName}</b> 被叛变波及阵亡`,'kill');
        if(G.stats.unitsLost[t.faction]!==undefined)G.stats.unitsLost[t.faction]++;
      }
    }
  }

  disFlash.style.setProperty('--dc','#e74c3c');
  disFlash.classList.remove('on');
  void disFlash.offsetWidth;
  disFlash.classList.add('on');
  disBanner.style.setProperty('--dc','#e74c3c');
  dbIco.textContent='👁';
  dbName.textContent='卧 底 暴 露';
  dbDesc.textContent=`${factionName(oldFac)} 的 ${u.typeName} 原为 ${factionName(newFac)} 卧底`;
  disBanner.classList.add('on');
  setTimeout(()=>disBanner.classList.remove('on'),3200);

  log(`👁 <b>${factionName(oldFac)}</b> 的 ${u.typeName} 暴露身份——真实阵营为 <b style="color:${factionColor(newFac)}">${factionName(newFac)}</b>，对周围造成 ${hit*25} 伤害！`,'spy');

}

function startTurn(fac){
  G.current=fac;G.busy=false;
  clearSelection();
  if(MAJOR_FACTIONS.includes(fac)){updateZYObservation(fac);maybeAIRequestZY(fac);}
  G.units.forEach(u=>{
    if(u.hp<=0)return;
    if(u.isBoss||u.isMinion||(u.isMXY&&fac!=='MXY'))return;
    u.moved=false;u.acted=false;u.buff=1;u.movBonus=0;
    u.flagBuff=false;u.rageBuff=false;u.shieldWall=false;u.zyBuff=false;
    u.slow=false;u.invisible=false;
    if(u.skillCd>0)u.skillCd--;
    u.energy=Math.min(100,u.energy+25);
    const terr=terrainAt(u.x,u.y);
    if(terr.heal>0&&u.hp<u.maxHp){
      const heal=terr.heal;
      u.hp=Math.min(u.maxHp,u.hp+heal);
      float(u,heal,'heal');
      if(MAP[u.y][u.x]===5)healFX(u.x,u.y);
    }
  });
  const fk=FACTIONS[fac];
  if(fac==='MXY'){
    const m=mxyAlive();
    if(!m){checkMXYDefeat();checkWin();return;}
    if(m.envoy){m.envoy=false;G.mxy.lastText='外交护行结束';}
    if((G.mxy.balanceCd||0)>0)G.mxy.balanceCd--;updateMXYBalanceExpiry();
    if(m.hp<m.maxHp){
      const heal=Math.min(12,m.maxHp-m.hp);
      m.hp+=heal;float(m,heal,'heal');healFX(m.x,m.y);
      log(`♛ <b>独行之主</b>：MXY 回复 ${heal} 兵力`,'skill');
    }
    const ev=currentMXYEvent();
    recordMXYMeet(m);evaluateMXYEvent(m);if(G.over)return;
  }
  if(fk.skill.type==='regen'){
    let healed=0;
    G.units.forEach(u=>{
      if(u.faction===fac&&u.hp>0&&u.hp<u.maxHp){
        const heal=12;
        u.hp=Math.min(u.maxHp,u.hp+heal);
        float(u,heal,'heal');healed++;
      }
    });
    if(healed>0)log(`<span style="color:${fk.color}">${fk.name}</span> 发动 <b>开源共振</b>`,'skill');
  }
  const lords=G.units.filter(u=>u.faction===fac&&u.hp>0&&u.type==='lord');
  lords.forEach(l=>{
    G.units.forEach(u=>{
      if(u.faction===fac&&u.hp>0&&u.id!==l.id&&dist(l,u)===1){
        if(u.hp<u.maxHp){
          const heal=8;
          u.hp=Math.min(u.maxHp,u.hp+heal);
          float(u,heal,'heal');
        }
      }
    });
  });
  if(MAJOR_FACTIONS.includes(fac)){
    const inc=computeIncome(fac);
    G.income[fac]=inc;G.gold[fac]+=inc;
    log(`<span style="color:${fk.color}">${fac}</span> 收入 +${inc} 金`,'sys');
    drawCard(fac);
  } else {G.income.MXY=0;G.gold.MXY=0;}

  // 每个完整回合仅在第一行动势力开始时判定一次随机战局事件 / NPC领主是否入局
  if(fac===G.order[0]){maybeTriggerBattleEvent();maybeSpawnMXY();}
  updateRoleProgress(fac);if(fac==='GYC')updateRebelEndgame();if(G.over)return;
  if(MAJOR_FACTIONS.includes(fac)){checkClinicalUnlocks(fac);applyClinicalSupport(fac);}

  if(G.disaster){tickDisaster();}else{maybeTriggerDisaster();}

  maybeStartShrink();

    applyShrinkDamage();

  }

  if(fac===G.player){}else{G.busy=true;renderHUD();setTimeout(()=>aiTurn(fac),620);}

function endTurn(){
  if(G.over)return;
  const idx=G.order.indexOf(G.current);
  let next=(idx+1)%G.order.length;
  let guard=0;
  while(guard++<4){
    const nf=G.order[next];
    if(G.units.some(u=>u.faction===nf&&u.hp>0))break;
    next=(next+1)%G.order.length;
  }
  const nf=G.order[next];
  if(next<=idx){
    G.round++;
    const afterSpecialTurns=()=>{G.busy=false;if(G.over){renderHUD();return;}startTurn(nf);};
    const runMXY=()=>{if(G.player!=='MXY'&&mxyAlive()&&!G.mxy.won){G.busy=true;renderHUD();mxyTurn(afterSpecialTurns);}else afterSpecialTurns();};
    if(G.bossActive && !G.bossDefeated){G.busy=true;renderHUD();setTimeout(()=>{bossTurn(()=>{if(G.over){G.busy=false;renderHUD();return;}runMXY();});},400);return;}
    if(G.player!=='MXY'&&mxyAlive()&&!G.mxy.won){G.busy=true;renderHUD();setTimeout(()=>mxyTurn(afterSpecialTurns),320);return;}
  }
  startTurn(nf);
}

function aiTurn(fac){
  const myUnits=G.units.filter(u=>u.faction===fac&&u.hp>0);
  let i=0;
  function step(){
    if(G.over){G.busy=false;renderHUD();return;}
    if(i>=myUnits.length){G.busy=false;renderHUD();setTimeout(endTurn,400);return;}
    const u=myUnits[i++];
    if(u.hp<=0){step();return;}
    let targets=alive().filter(e=>e.faction!==fac);
    if(G.bossActive && !G.bossDefeated){
      const bossUnits=targets.filter(e=>e.isBoss||e.isMinion);
      if(bossUnits.length) targets=bossUnits;
    }
    if(!targets.length){G.busy=false;renderHUD();setTimeout(endTurn,400);return;}

    if((G.hand[fac]||[]).length>0 && Math.random()<0.3){
      const hand=G.hand[fac];
      const idx=hand.findIndex(c=>c.target==='unit');
      if(idx>=0){
        const c=hand[idx];
        const target=myUnits.filter(x=>x.hp<x.maxHp*0.6).sort((a,b)=>a.hp-b.hp)[0]||u;
        G.selected=target;
        const ok=c.use(target);
        G.selected=null;
        if(ok){
          hand.splice(idx,1);
          log(`<b style="color:${factionColor(fac)}">${factionName(fac)}</b> 使用了功能卡 <b>${c.name}</b>`,'card');
          renderUnits();renderHUD();
          setTimeout(step,500);
          return;
        }
      }
    }

    if(canUseSkill(u)&&Math.random()<0.4){
      let use=false;
      if(u.type==='lord'){
