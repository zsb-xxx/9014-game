function calcAtk(u){
  let atk=(u.isBoss?bossEffectiveAtk(u):u.atk)*(u.buff||1);
  const f=FACTIONS[u.faction];
  if(f&&f.skill.type==='rage'&&u.hp/u.maxHp<0.5)atk*=1.4;
  if(u.rageBuff)atk*=1.5;
  atk*=terrainAt(u.x,u.y).atkMod;
  if(u.flagBuff)atk*=1.2;
  if(G.disaster){
    if(G.disaster.key==='bloodmoon')atk*=1.3;
    if(G.disaster.key==='zy-surge'&&onZY(u.x,u.y))atk*=1.2;
  }
  return Math.round(atk);
}
function calcRange(u){
  const t=TYPES[u.type];if(!t)return 1;
  let r=t.range;
  const f=FACTIONS[u.faction];
  if(f&&f.skill.type==='tactics'&&u.type==='sage')r+=1;
  r+=(terrainAt(u.x,u.y).rngMod||0);
  if(G.disaster&&G.disaster.key==='storm')r-=1;
  return Math.max(1,r);
}
function calcMov(u){
  let mov=u.mov+(u.movBonus||0);
  const f=FACTIONS[u.faction];
  if(f&&f.skill.type==='tactics')mov+=1;
  if(G.disaster&&G.disaster.key==='blizzard'&&!onZY(u.x,u.y))mov-=1;
  if(u.slow)mov-=1;
  return Math.max(1,mov);
}
function isRaging(u){

}
function canUseSkill(u){
  const sk=SKILLS[u.type];
  if(!sk)return false;
  if(u.type==='mxy_lord'&&currentMXYEvent()?.key!=='meet')return false;
  return u.skillCd===0&&u.energy>=sk.cost&&!u.acted&&u.hp>0;
}

function inPattern(u,x,y){
  if(x===u.x&&y===u.y)return false;
  const dx=Math.abs(x-u.x),dy=Math.abs(y-u.y);
  const man=dx+dy,che=Math.max(dx,dy);
  const t=TYPES[u.type];if(!t)return false;
  const r=calcRange(u);
  switch(t.pattern){
    case 'ring':return che===1;
    case 'cross':return man===1;
    case 'diagonal':return dx===1&&dy===1;
    case 'bossring':return che<=r&&che>0;
    case 'line':
      if(dx!==0&&dy!==0)return false;
      if(man<1||man>r)return false;
      return hasLOS(u.x,u.y,x,y);
    default:return man<=r;
  }
}
function hasLOS(x0,y0,x1,y1){
  if(x0!==x1&&y0!==y1)return false;
  const dx=Math.sign(x1-x0),dy=Math.sign(y1-y0);
  let cx=x0+dx,cy=y0+dy;
  while(cx!==x1||cy!==y1){
    if(!isPassableTile(cx,cy))return false;
    cx+=dx;cy+=dy;
  }
  return true;
}
function computeAtk(u){
  const res=[];
  for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++)if(inPattern(u,x,y))res.push([x,y]);
  return res;
}
